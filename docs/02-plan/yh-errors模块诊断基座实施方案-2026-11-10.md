# errors 诊断基座模块实施方案

> **执行约定：** 按任务逐条实施，每个任务结束必须 `just dev` + `just test` 全绿
> 并提交（commit 只用 pathspec 带本任务文件）；Task 1 与 Task 6 额外跑
> `just selfhost-native`。写 `.yh` 前已读 [yh-编码规范](../03-design/yh-编码规范.md)。

**目标：** 把编译器统一错误输出收敛到新的叶子模块 `errors`，其他模块一律
`import errors` 报错；顺带修复 CI 四端门禁红（孤儿 errors 模块缺产物）。

**架构：** `errors` 不 import 任何模块，成为比 `support` 更底层的基座：
`DiagCtx` 状态与 `fail`/`fail_path`/`err_lc`/`warn_lc` 四个诊断入口从
`support` 原样迁入（渲染逻辑逐字节不变），`support` 及其他模块经委托过渡后
全部改为直接调用 `errors`。`write`/`malloc`/`exit` 是 prelude 内置，叶子模块
无需任何 import 即可写 stderr。

**技术栈：** yh 语言、just 门禁、tools/compiler-tests.py（模块内 yh test）、
tests/selfhost 固定点门。

---

## 背景与根因

- CI run 42756269175（merge `13d3bd4`）四端同挂：
  `FAIL cc2 产物：cc2 缺少产物：iface/yh-errors.s.ct, iface/yh-errors.s.i, llvm/yh-errors.ll`。
- 根因：`8a398d0` 新增 `compiler/errors/api.yh`（`module errors` 骨架），但全
  编译器无一模块 import 它 → 按需编译不产生其产物；而
  `tools/selfhost_snapshot_gate.py:64` 按源码树扫描**所有** module 声明计算
  预期产物集 → cc2 缺三项 → `just selfhost-native` 红。
- 骨架的 `Status`/`Error`/空函数是 Result 风格，与编译器内部「int 返回码 +
  就地诊断」约定（编码规范 §8）不符，且空实现违反「每个 pub 真的被使用」。

## 决策与假设（已经用户确认）

1. yh 无继承机制，「其他模块继承该模块输出错误」落地为 **import 依赖**。
2. **叶子基座**（用户选定）：errors 不 import 任何模块；DiagCtx 与四个诊断
   入口从 support 迁入；support 退位为通用工具层。
3. 删除骨架的 `Status`/`Error`/`ok()`/`error()`/`fail()` 空壳，不做 Result
   风格错误值。
4. 诊断输出文本**逐字节不变**：渲染逻辑原样搬迁，cli-golden 与 fail 语料
   不需要更新。
5. 迁移期 support 保留五个**临时委托**函数保证每个任务都绿；Task 6 必须
   删除，不允许长期两套入口。

## 明确不做

- 不改诊断文本、不加错误码/分级/结构化错误（todo 长期项「错误提示太简陋」
  是后续工作）。
- 不收编约 20 处 `support.write_all(2, ...)` 直写点（link 的 lld 诊断透传、
  build 进度输出等）：它们不是 `yh: ` 前缀的诊断入口，文本各异，收编列入
  后续 todo。
- 不动 `support` 的通用缓冲函数（`append_str`/`write_all` 等仍 pub，105+
  文件在用）；errors 自带一份最小私有渲染辅助（见「风险」）。

`compiler/errors/api.yh` 完整内容：

```yh
module errors

// ============================================================================
// api.yh —— errors 模块公开面：编译器统一错误输出基座
//
// 本模块是编译器最底层（不 import 任何模块）：DiagCtx 状态、带位置的
// err_lc/warn_lc 与不带位置的 fail/fail_path 统一收在这里，其他模块一律
// import errors 报错。明确不做：不分配、不格式化语义错误码、不提供 Result
// 风格错误值（编译器内部约定 int 返回码 + 就地诊断，见编码规范 §8）。
// ============================================================================

// DiagCtx：诊断的进程级开关与计数。
// spec>0 表示推测解析，诊断静默且不计数；errors 是已报告错误数；
// line_bias 是 .s.ct 来源映射的行号偏移；ct_* 是编译期 for 展开轨迹。
pub struct DiagCtx {
    spec int // 推测解析深度（>0 静默）
    errors int // 已报告错误数
    line_bias int // 诊断行号映射偏移
    ct_depth int // 当前编译期 for 展开深度
    ct_values [32]int // 外层到内层的展开值
}

// g_diag：进程内唯一诊断状态（零初始化即全部默认值）。
let g_diag DiagCtx

// g_path：当前源文件路径（借用，不拥有）。由 support.set_source_path 在
// 路径缓冲定型后登记，两次登记之间调用方保证有效；err_lc/warn_lc 只读。
let g_path *char

// diag：诊断上下文窄口；返回进程单例指针。
pub fn diag() *DiagCtx {
    return &g_diag
}

// set_path：登记当前源文件路径（借用，不拷贝）。path 必须是 NUL 串且在
// 下次登记前保持有效；首次带位置诊断前必须登记（由 support.set_source_path
// 保证），本函数不判空、不分配。
pub fn set_path(path *char) void {
    g_path = path
}

// fail：报告一条不带位置的编译器级错误（打开失败、内存不足等），返回 -1。
// 写 stderr；不分配、不检查 msg 非空（调用方保证是静态/池内 NUL 串）。
pub fn fail(msg *char) int {
    let buf [512]u8
    let n int = 0
    n = append_str_cap(buf, 512, n, "yh: ")
    n = append_str_cap(buf, 512, n, msg)
    n = append_byte_cap(buf, 512, n, '\n')
    write_all(2, buf, n)
    return -1
}

// fail_path：报告与源文件路径相关的错误（如无法打开），返回 -1。
// 写 stderr；不分配、不检查 path 非空（调用方保证是 NUL 串）。
pub fn fail_path(msg *char, path *char) int {
    let buf [512]u8
    let n int = 0
    n = append_str_cap(buf, 512, n, "yh: ")
    n = append_str_cap(buf, 512, n, msg)
    n = append_str_cap(buf, 512, n, path)
    n = append_byte_cap(buf, 512, n, '\n')
    write_all(2, buf, n)
    return -1
}

// err_lc：报告一条带位置的错误（1 基行列），计一次错误并返回 -1。
// 推测解析期间静默：不打印、不计数，仅以返回值示意失败。
// 调用前必须经 set_path 登记路径；行号叠加 diag().line_bias。
pub fn err_lc(line int, col int, msg *char) int {
    if (diag().spec > 0) {
        return -1
    }
    diag().errors += 1
    let buf [2048]u8
    let limit int = 2047
    let n int = 0
    n = append_str_limit(buf, n, limit, g_path)
    n = append_byte(buf, n, ':')
    n = append_int(buf, n, line + diag().line_bias)
    n = append_byte(buf, n, ':')
    n = append_int(buf, n, col)
    n = append_str_limit(buf, n, limit, ": 错误: ")
    n = append_str_limit(buf, n, limit, msg)
    n = append_ct_expand_trace(buf, n, limit)
    if (n < limit) { n = append_byte(buf, n, '\n') }
    else { buf[limit] = '\n'; n = limit + 1 }
    write_all(2, buf, n)
    return -1
}

// warn_lc：报告一条带位置的警告，不增加错误数；推测解析期间保持静默。
// 返回 0 成功、-1 表示写 stderr 失败；调用前必须经 set_path 登记路径。
pub fn warn_lc(line int, col int, msg *char) int {
    if (diag().spec > 0) { return 0 }
    let buf [2048]u8
    let limit int = 2047
    let n int = 0
    n = append_str_limit(buf, n, limit, g_path)
    n = append_byte(buf, n, ':')
    n = append_int(buf, n, line + diag().line_bias)
    n = append_byte(buf, n, ':')
    n = append_int(buf, n, col)
    n = append_str_limit(buf, n, limit, ": 警告: ")
    n = append_str_limit(buf, n, limit, msg)
    n = append_ct_expand_trace(buf, n, limit)
    if (n < limit) { n = append_byte(buf, n, '\n') }
    else { buf[limit] = '\n'; n = limit + 1 }
    return write_all(2, &buf[0], n)
}
```

`compiler/errors/render.yh` 完整内容（前六个函数与 support/api.yh 同名函数
逐字节同形，只是改私有；append_ct_expand_trace 迁自 support/diag.yh）：

```yh
module errors

// ============================================================================
// render.yh —— 诊断消息的私有渲染辅助
//
// 与 support 的通用缓冲函数同形，但本模块不 import support（叶子基座），
// 故自带一份最小私有实现：只服务 api.yh 的四个诊断入口，不做通用工具。
// 所有函数不分配；容量由调用方的栈缓冲保证。
// ============================================================================

// append_byte：向 buf 的 n 处写入一个字节，返回新的写入位置。
// 仅用于栈上小缓冲，不做容量检查——调用方保证空间。
fn append_byte(buf *u8, n int, b u8) int {
    buf[n] = b
    return n + 1
}

// append_int：向 buf 追加十进制整数（含负号），返回新位置。
// 行号/列号规模远小于 2^31，不处理 INT_MIN 取负溢出。
fn append_int(buf *u8, n int, v int) int {
    if (v < 0) {
        buf[n] = '-'
        n += 1
        v = -v
    }
    // 逆序取位到 tmp，再反向拷入 buf
    let tmp [12]u8
    let k int = 0
    if (v == 0) {
        tmp[0] = '0'
        k = 1
    }
    for v > 0 {
        tmp[k] = ((v % 10) + 48).to_type(u8)
        k += 1
        v = v / 10
    }
    for k > 0 {
        k -= 1
        buf[n] = tmp[k]
        n += 1
    }
    return n
}

// append_str_limit：向 buf 追加字符串，最多写到 limit 之前，返回新位置。
// 用于包含源代码消息的诊断拼装，避免长输入写出固定缓冲区边界。
fn append_str_limit(buf *u8, n int, limit int, s *char) int {
    let i int = 0
    for s[i] != 0 && n < limit {
        buf[n] = s[i].to_type(u8)
        n += 1
        i += 1
    }
    return n
}

// append_str_cap：容量受限追加字符串，返回新的长度（写满即止）。
fn append_str_cap(buf *u8, cap int, n int, s *char) int {
    let i int = 0
    for n < cap - 1 && s[i] != 0 {
        buf[n] = s[i].to_type(u8)
        n += 1
        i += 1
    }
    buf[n] = 0
    return n
}

// append_byte_cap：容量受限追加单字节，返回新的长度（写满即止）。
fn append_byte_cap(buf *u8, cap int, n int, b u8) int {
    if (n >= cap - 1) { return n }
    buf[n] = b
    n += 1
    buf[n] = 0
    return n
}

// write_all：把 buf 前 n 个字节全部写到 fd。
// write 可能短写，循环推进直到写完；负返回视为错误。
// 返回 0 成功、-1 表示 write 失败（不重试 EINTR，调用方按编译失败处理）。
fn write_all(fd int, buf *u8, n int) int {
    let done int = 0
    for done < n {
        let w i64 = write(fd, buf + done, (n - done).to_type(u64))
        if (w <= 0) {
            return -1
        }
        done += w.to_type(int)
    }
    return 0
}

// append_ct_expand_trace：为展开体内诊断追加外层到内层的编译期循环值。
fn append_ct_expand_trace(buf *u8, n int, limit int) int {
    if (diag().ct_depth <= 0 || n >= limit - 600) { return n }
    n = append_str_limit(buf, n, limit, " [编译期展开值: ")
    let i int = 0
    for i < diag().ct_depth {
        if (i > 0) { n = append_str_limit(buf, n, limit, " -> ") }
        n = append_int(buf, n, diag().ct_values[i])
        i += 1
    }
    return append_str_limit(buf, n, limit, "]")
}
```

## 风险与权衡

1. **渲染辅助重复 ~110 行**（errors/render.yh 私有 vs support 通用版）：
   有意为之，保住 errors 零 import 的叶子地位；两处同形函数由固定点门与
   语料门共同约束行为。日后若出现第三个消费者再考虑下沉。
2. **路径槽位不变量（2026-11-10 修正）**：errors 持路径**字段的地址**而非
   路径快照——`set_path_slot(&g_support.src.path)` 在会话初始化登记一次，
   err_lc/warn_lc 调用时刻读 `*g_path_slot`。14 处直写 src_ctx().path 的
   换源点（含 ct_diag 的保存/恢复）自动可见，缓冲搬迁也不会产生悬空快照。
   约束：登记必须先于首次带位置诊断；g_support 单例地址进程内稳定。
   初版快照设计（g_path + set_path）已在 just test 中证伪并废弃。
3. **固定点门**：cc2 由 0.0.14 种子编译新树，errors 只用种子已支持的既有
   语法（结构体/数组字段/指针全局零初始化/prelude write）；Task 1 的
   `just selfhost-native` 是第一道实证。
4. **迁移期双入口**：support 五个委托是临时脚手架，Task 6 删除前每个任务
   都保持全绿；不允许把委托留到方案之外。
5. **`let g_path_slot **char` 零初始化全局指针**：若种子对该形态 lowering 有
   缺口，Task 1 Step 4 会立即暴露；退路是把槽位收进一个 struct 单例
   （与 g_support 同形）。
