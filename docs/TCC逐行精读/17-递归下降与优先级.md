# 17 · 递归下降与优先级：函数套娃

> **一句话**：C 的运算符优先级，在 TCC 里不靠"优先级表"，而靠**函数的嵌套深度**——
> 越靠里调用的函数，结合得越紧（优先级越高）。
>
> 对应源码：`src/core/tccgen.c:5286-5687`（一整套 `expr_*`）、
> `src/core/tccgen.c:5678`（`gexpr` 顶层入口）、
> `src/core/tccgen.c:4594`（最底层 `unary`，第 18/19 篇）。

---

## 第 0 站：C 的优先级表 vs 函数套娃

C 的优先级从低到高大致是：

```
逗号
  ?:              （三元）
    ||
      &&
        |
          ^
            &
              == !=
                < <= > >=
                  << >>
                    + -
                      * / %
                        一元（- ! ~ * & ++ -- sizeof 强制转换）
                          后缀（[] . -> () ++ --）
```

教科书会读一张表。TCC 的做法是写一串函数，**每个函数只处理自己这一层**：

```c
/* src/core/tccgen.c:5677-5687 */
ST_FUNC void gexpr(void)          /* 最顶层：逗号 */
{
    while (1) {
        expr_eq();
        if (tok != ',')
            break;
        vpop();                   /* 逗号：扔掉左边，继续右边 */
        next();
    }
}
```

**调用链**（越往下优先级越高）：

```
gexpr
 └ expr_eq        (= += -= ...，右结合)
    └ expr_cond   (?:)
       └ expr_lor (||)
          └ expr_land (&&)
             └ expr_or  (|)
                └ expr_xor (^)
                   └ expr_and (&)
                      └ expr_cmpeq (== !=)
                         └ expr_cmp (< <= > >=)
                            └ expr_shift (<< >>)
                               └ expr_sum (+ -)
                                  └ expr_prod (* / %)
                                     └ unary
                                        └ postfix
```

**优先级 = 在调用链里的深浅。** 越深，越先被解析、越先结合。

## 第 1 站：最标准的一层 —— `expr_prod` / `expr_sum`

```c
/* src/core/tccgen.c:5286-5310 */
ST_FUNC void expr_prod(void)          /* 乘除取余 */
{
    int t;
    unary();                          /* 先吃一个"最底层的项" */
    while (tok == '*' || tok == '/' || tok == '%') {
        t = tok;
        next();                       /* 吃掉运算符 */
        unary();                      /* 再吃一个项 */
        gen_op(t);                    /* ★ 栈顶两项归约 */
    }
}

ST_FUNC void expr_sum(void)           /* 加减 */
{
    int t;
    expr_prod();                      /* 先吃一个"乘积"（注意！调的是更深的层） */
    while (tok == '+' || tok == '-') {
        t = tok;
        next();
        expr_prod();                  /* 再吃一个乘积 */
        gen_op(t);
    }
}
```

看懂 `expr_sum` 调 `expr_prod`（而不是反过来）就懂了一切：

- 解析 `a + b * c`：
  1. `expr_sum` 调 `expr_prod` → 吃到 `a`；
  2. 看到 `+`，吃掉，再调 `expr_prod` → 吃到 `b`，看到 `*`，再吃 `c`，
     `gen_op('*')` 归约成一项；没有更多 `*`，返回；
  3. `gen_op('+')` 归约 `a` 和 `b*c`。
- **乘法的结合发生在更深的 `expr_prod` 里，所以它先算。**

所有层结构一模一样，只换"调谁"和"认哪些运算符"：

```c
/* src/core/tccgen.c:5312-5380（连排）*/
static void expr_shift(void) { expr_sum();   while (tok==TOK_SHL||tok==TOK_SAR) { t=tok; next(); expr_sum();   gen_op(t);} }
static void expr_cmp(void)   { expr_shift(); while (<比较运算符>)              { t=tok; next(); expr_shift(); gen_op(t);} }
static void expr_cmpeq(void) { expr_cmp();   while (tok==TOK_EQ||tok==TOK_NE)  { t=tok; next(); expr_cmp();   gen_op(t);} }
static void expr_and(void)   { expr_cmpeq(); while (tok=='&')                  { next(); expr_cmpeq(); gen_op('&');} }
static void expr_xor(void)   { expr_and();   while (tok=='^')                  { next(); expr_and();   gen_op('^');} }
static void expr_or(void)    { expr_xor();   while (tok=='|')                  { next(); expr_xor();   gen_op('|');} }
```

> **注意左结合**：`a-b-c` 在 `expr_sum` 里循环处理成 `(a-b)-c`，
> 因为每读到一个 `-` 就立刻 `gen_op('-')` 把左边归约掉，再读右边。
> 而右结合的赋值在 `expr_eq` 里用**递归**（`expr_eq()` 调自己），所以是 `a=(b=c)`。

## 第 2 站：赋值的右结合 —— `expr_eq`（`tccgen.c:5655`）

```c
/* src/core/tccgen.c:5655-5676 */
static void expr_eq(void)
{
    int t;

    expr_cond();                        /* 先吃一个三元表达式 */
    if (tok == '=' ||
        (tok >= TOK_A_MOD && tok <= TOK_A_DIV) ||
        tok == TOK_A_XOR || tok == TOK_A_OR ||
        tok == TOK_A_SHL || tok == TOK_A_SAR) {
        test_lvalue();                  /* ★ 左边必须是"位置" */
        t = tok;
        next();
        if (t == '=') {
            expr_eq();                  /* ★ 递归调用自己：右结合！ */
        } else {
            vdup();                     /* a += b 等价于 a = a + b */
            expr_eq();
            gen_op(t & 0x7f);           /* 复合赋值：取运算符 */
        }
        vstore();                       /* ★ 写回左值（第 24 篇） */
    }
}
```

关键区别：

- **普通赋值 `=`**：直接 `expr_eq()` 递归右操作数 → `a = (b = c)`。
- **复合赋值 `+=`**：`vdup()` 复制左值、算 `a op b`、再 `vstore` 写回。
  `t & 0x7f` 把 `TOK_A_ADD` 变成 `'+'`（复合赋值的 token 编号设计成
  "基础运算符编号 + 高位标志"，一按位与就还原）。

## 第 3 站：三元 `?:`（`tccgen.c:5480`）

`expr_cond` 比别的层复杂，因为两分支都要生成代码。骨架：

```c
/* src/core/tccgen.c:5480-5537（节选）*/
static void expr_cond(void)
{
    int tt, ..., c, g;
    SValue sv;

    expr_lor();                       /* 先吃条件 */
    if (tok == '?') {
        next();
        c = condition_3way();         /* 能编译期定真假吗？-1=不知道 */
        ...
        if (c < 0) {
            gv(rc);                   /* 条件物化到寄存器 */
            save_regs(1);
            tt = gvtst(1, 0);         /* ★ 生成"条件为假就跳" */
        }
        ...
        gexpr();                      /* 生成"真"分支 */
        ...
        skip(':');
        u = gjmp(0);                  /* 真分支结束，跳到汇合点 */
        gsym(tt);                     /* 回填"假则跳"到假分支开始 */
        ...
        expr_cond();                  /* 生成"假"分支（右结合） */
        ...
        tt = gjmp(0);
        gsym(u);                      /* 回填真分支的跳转 */
        ...
        gv(rc); move_reg(...);        /* 让两分支结果落到同一寄存器 */
    }
}
```

`condition_3way`（`:5467`）是"能算就算"的优化：

```c
/* src/core/tccgen.c:5467-5478 */
static int condition_3way(void)
{
    int c = -1;
    if ((vtop->r & (VT_VALMASK | VT_LVAL)) == VT_CONST &&
        (!(vtop->r & VT_SYM) || !vtop->sym->a.weak)) {
        vdup();
        gen_cast_s(VT_BOOL);
        c = vtop->c.i;                /* 条件是常量：直接得真假 */
        vpop();
    }
    return c;
}
```

于是 `1 ? a : b` 在编译期就知道只该生成 `a`，另一分支用 `nocode_wanted` 跳过。

## 第 4 站：短路 `&&` / `||`（`tccgen.c:5382/5423`）

`&&` 和 `||` **不能简单地两边都生成**——左边决定右边算不算。看 `expr_land`：

```c
/* src/core/tccgen.c:5382-5421（节选）*/
static void expr_land(void)
{
    expr_or();
    if (tok == TOK_LAND) {
        int t = 0;
        for(;;) {
            if (vtop 是常量) {
                gen_cast_s(VT_BOOL);
                if (vtop->c.i) { vpop(); }          /* 常量真：看右边 */
                else {
                    nocode_wanted++;                 /* 常量假：右边全跳过 */
                    while (tok == TOK_LAND) { next(); expr_or(); vpop(); }
                    nocode_wanted--;
                    ...
                    break;
                }
            } else {
                if (!t) save_regs(1);
                t = gvtst(1, t);   /* ★ 左边假就跳走（1 = 反转测试） */
            }
            if (tok != TOK_LAND) {
                if (t) vseti(VT_JMPI, t);  /* 左边假 → 整个结果假 */
                else   vpushi(1);
                break;
            }
            next();
            expr_or();             /* 右边 */
        }
    }
}
```

核心：`gvtst(1, t)` 生成"**条件为假就跳到终点**"，并把跳转链累积在 `t` 里。
链尾用 `vseti(VT_JMPI, t)` 压一张"跳转链"便利贴（第 12 篇的 `VT_JMP`）。
真正的回填发生在你**使用这个布尔值**时（`if`/`while` 会调 `gvtst` 消化它）。

`expr_lor` 是对偶的：`gvtst(0, t)` 表示"条件为真就跳走"。

**这种"先造跳转链、用到才回填"的做法，让短路求值不需要额外数据结构和额外遍历。**

## 第 5 站：只吃常量 —— 常量表达式入口

```c
/* src/core/tccgen.c:5689-5721 */
static void expr_const1(void)     /* 解析常量表达式 */
{
    const_wanted++;               /* ★ 要求常量 */
    nocode_wanted++;              /* ★ 不生成代码 */
    expr_cond();
    nocode_wanted--;
    const_wanted--;
}

static inline int64_t expr_const64(void)
{
    int64_t c;
    expr_const1();
    if ((vtop->r & (VT_VALMASK|VT_LVAL|VT_SYM)) != VT_CONST)
        expect("constant expression");     /* 不是常量就报错 */
    c = vtop->c.i;
    vpop();
    return c;
}

ST_FUNC int expr_const(void)
{
    int c;
    int64_t wc = expr_const64();
    c = wc;
    if (c != wc && (unsigned)c != wc)
        tcc_error("constant exceeds 32 bit");
    return c;
}
```

**同一个表达式解析器，改两个开关就能当"常量求值器"用**：

- `const_wanted++` → 遇到需要变量的地方报错；
- `nocode_wanted++` → 不生成机器码。

`#if` 表达式（第 08 篇）、数组大小、`case` 标签、枚举值、全局初始化值，
全走这条入口。**这就是复用的威力。**

## 【把 `a + b * c > d && e` 走一遍】

调用链深度决定了结合顺序：

```
expr_land
 ├ expr_or  ← 左
 │  └ expr_xor
 │     └ expr_and
 │        └ expr_cmpeq
 │           └ expr_cmp
 │              ├ expr_shift→...→expr_sum
 │              │   ├ expr_prod→unary  ← a
 │              │   └ expr_prod
 │              │       ├ unary ← b
 │              │       └ gen_op('*')  ← b*c
 │              │   └ gen_op('+')      ← a+(b*c)
 │              └ expr_shift→...→unary ← d
 │              └ gen_op('>')          ← (a+b*c)>d
 ├ gvtst(1,...)  ← 左边假就跳
 └ expr_or ← e
```

结果：`(a + (b*c)) > d` 先算，再和 `e` 短路与。

## 【常见卡点】

- **优先级靠嵌套，不靠表**：函数越深，优先级越高。
- **左结合靠"读到运算符立刻归约"**，右结合靠"递归自己"。
- **`&&`/`||` 造跳转链，用到才回填**，不是两边都生成。
- **常量表达式复用同一套解析器**，靠 `const_wanted`/`nocode_wanted` 开关。
