## 第 1 站：压栈（`tccgen.c:630-761`）

```c
/* src/core/tccgen.c:630-661（再看一遍，注意"物化"）*/
static void vsetc(CType *type, int r, CValue *vc)
{
    int v;
    if (vtop >= vstack + (VSTACK_SIZE - 1))
        tcc_error("memory full (vstack)");
    if (vtop >= vstack && !nocode_wanted) {
        v = vtop->r & VT_VALMASK;
        if (v == VT_CMP || (v & ~1) == VT_JMP)
            gv(RC_INT);              /* ★ 栈顶是 CMP/JMP：先物化 */
    }
    vtop++;
    vtop->type = *type;
    vtop->r = r;
    vtop->r2 = VT_CONST;
    vtop->c = *vc;
    vtop->sym = NULL;
}
```

其余压栈函数都是它的包装：

```c
/* src/core/tccgen.c:695-756（节选）*/
ST_FUNC void vpush(CType *type)               { vset(type, VT_CONST, 0); }
ST_FUNC void vpushi(int v)                    { CValue c; c.i = v; vsetc(&int_type, VT_CONST, &c); }
static void vpushs(addr_t v)                  { CValue c; c.i = v; vsetc(&size_type, VT_CONST, &c); }
ST_FUNC void vpush64(int ty, unsigned long long v) { ... vsetc(&ctype, VT_CONST, &c); }
ST_FUNC void vset(CType *type, int r, int v)  { CValue c; c.i = v; vsetc(type, r, &c); }
ST_FUNC void vpushv(SValue *v)                { vtop++; *vtop = *v; }
static void vdup(void)                        { vpushv(vtop); }
```

**`vpushv` 值得单独看**：它直接 `*vtop = *v`，把一整张便利贴（含 `sym` 欠条、
`r2` 第二寄存器）原样复制。`vdup()` 就是复制栈顶。

## 第 2 站：弹栈 —— `vpop` 的资源清理（`tccgen.c:678`）

```c
/* src/core/tccgen.c:677-693 */
ST_FUNC void vpop(void)
{
    int v;
    v = vtop->r & VT_VALMASK;
#if defined(TCC_TARGET_I386) || defined(TCC_TARGET_X86_64)
    if (v == TREG_ST0) {
        o(0xd8dd);              /* fstp %st(0)：弹掉 x87 浮点栈 */
    } else
#endif
    if (v == VT_JMP || v == VT_JMPI) {
        gsym(vtop->c.i);        /* 回填未完成的跳转链 */
    }
    vtop--;
}
```

**`vpop` 不是单纯的指针减一**：

- 栈顶是 x87 浮点寄存器 `ST0` → 发一条 `fstp` 清浮点栈（否则会溢出）；
- 栈顶是跳转链 → `gsym` 回填（`&&`/`||` 的链尾）。

**所以"弹栈"有时候是要产码的**，速查表里标了"可能"。

## 第 3 站：交换与旋转（`tccgen.c:663-797`）

```c
/* src/core/tccgen.c:663-675 */
ST_FUNC void vswap(void)
{
    SValue tmp;
    if (vtop >= vstack && !nocode_wanted) {
        int v = vtop->r & VT_VALMASK;
        if (v == VT_CMP || (v & ~1) == VT_JMP)
            gv(RC_INT);         /* 同样先物化 */
    }
    tmp = vtop[0];
    vtop[0] = vtop[-1];
    vtop[-1] = tmp;
}
```

```c
/* src/core/tccgen.c:766-775 */
/* 把最上面 n 张转到底部： I1...In -> I2...In I1 [栈顶在右] */
ST_FUNC void vrotb(int n)
{
    int i;
    SValue tmp;
    tmp = vtop[-n + 1];
    for(i=-n+1;i!=0;i++)
        vtop[i] = vtop[i+1];
    vtop[0] = tmp;
}

/* 把 e 前面的 n 张往栈顶方向转：I1...In... -> In I1...I(n-1)... */
ST_FUNC void vrote(SValue *e, int n)
{
    int i;
    SValue tmp;
    tmp = *e;
    for(i = 0;i < n - 1; i++)
        e[-i] = e[-i - 1];
    e[-n + 1] = tmp;
}

ST_FUNC void vrott(int n) { vrote(vtop, n); }   /* 最上 n 张转到顶部 */
```

这三个用在**函数调用、三元、结构体比较**等"参数顺序要调"的地方。
例如函数调用时实参从左到右压栈，但后端要"从栈顶倒着装"，
中间会用到 `vrotb`/`vrott` 调整顺序。

**要点：旋转是纯数据搬运，不产码。**

## 第 4 站：左值相关 —— `test_lvalue` / `lvalue_type` / `gaddrof`

```c
/* src/core/tccgen.c:125-129 */
ST_FUNC void test_lvalue(void)
{
    if (!(vtop->r & VT_LVAL))
        expect("lvalue");
}
```

赋值左边、`&` 操作数必须是"位置"（`VT_LVAL`）。

```c
/* src/core/tccgen.c:4471-4485 */
ST_FUNC int lvalue_type(int t)
{
    int bt, r;
    r = VT_LVAL;                 /* 基本标志 */
    bt = t & VT_BTYPE;
    if (bt == VT_BYTE || bt == VT_BOOL)
        r |= VT_LVAL_BYTE;       /* 记住"这是 1 字节左值" */
    else if (bt == VT_SHORT)
        r |= VT_LVAL_SHORT;      /* 2 字节 */
    else
        return r;                /* 其它（int/指针/…）就是普通字 */
    if (t & VT_UNSIGNED)
        r |= VT_LVAL_UNSIGNED;   /* 有无符号也要记住 */
    return r;
}
```

**为什么左值要记字节大小？** 因为 `char c; c = 1;` 只该写 1 字节，
`load`/`store` 要根据 `VT_LVAL_BYTE/SHORT` 选 `movb`/`movw` 还是 `movl`。
（第 46 篇 `load`/`store` 会用到。）

```c
/* src/core/tccgen.c:1117-1126 */
ST_FUNC void gaddrof(void)
{
    vtop->r &= ~VT_LVAL;                 /* 去掉"左值"标志：从位置变成地址 */
    if ((vtop->r & VT_VALMASK) == VT_LLOCAL) {
        /* 如果它之前被溢出到栈上，恢复成偏移形式的地址 */
        vtop->r = (vtop->r & ~(VT_VALMASK | VT_LVAL_TYPE)) | VT_LOCAL | VT_LVAL;
    }
}
```

**`gaddrof` 只改标志位。** `&x` 和 `x` 在值栈上的差别就是"有没有 `VT_LVAL`"。

## 第 5 站：char/short 的延迟转换 —— `force_charshort_cast`（`tccgen.c:2417`）

`char`/`short` 运算前常要提升成 `int`，但直接转浪费指令。TCC 用 `VT_MUSTCAST`
延迟到真正需要值的时候：

```c
/* src/core/tccgen.c:2417-2448 */
static void force_charshort_cast(int t)
{
    int bits, dbt;
    if (STATIC_DATA_WANTED) return;
    dbt = t & VT_BTYPE;
    if (dbt == VT_BYTE) bits = 8;
    else                bits = 16;
    if (t & VT_UNSIGNED) {
        vpushi((1 << bits) - 1);
        gen_op('&');                 /* 无符号：与掩码 */
    } else {
        if ((vtop->type.t & VT_BTYPE) == VT_LLONG) bits = 64 - bits;
        else                                       bits = 32 - bits;
        vpushi(bits);
        gen_op(TOK_SHL);             /* 有符号：先左移 */
        vtop->type.t &= ~VT_UNSIGNED;
        vpushi(bits);
        gen_op(TOK_SAR);             /* 再算术右移 → 符号扩展 */
    }
}
```

**"左移再算术右移"是符号扩展的经典技巧**：
把低 8 位弄到最高位，再算术右移回来，高位就补成了符号位。

`gen_cast`（第 23 篇）开头遇到 `VT_MUSTCAST` 就调它。**这样避免了"赋值给 char 后立刻又读"的冗余转换。**

## 第 6 站：`lexpand`（仅 32 位平台，`tccgen.c:1488`）

32 位平台上 `long long` 要拆成两个 32 位：

```c
/* src/core/tccgen.c:1486-1507 */
static void lexpand(void)
{
    int u, v;
    u = vtop->type.t & (VT_DEFSIGN | VT_UNSIGNED);
    v = vtop->r & (VT_VALMASK | VT_LVAL);
    if (v == VT_CONST) {
        vdup();                       /* 复制一份 */
        vtop[0].c.i >>= 32;           /* 上面那份取高 32 位 */
    } else if (v == (VT_LVAL|VT_CONST) || v == (VT_LVAL|VT_LOCAL)) {
        vdup();
        vtop[0].c.i += 4;             /* 上面那份指向高 4 字节 */
    } else {
        gv(RC_INT);
        vdup();
        vtop[0].r = vtop[-1].r2;
        vtop[0].r2 = vtop[-1].r2 = VT_CONST;
    }
    vtop[0].type.t = vtop[-1].type.t = VT_INT | u;
}
```

**64 位的 x86-64 不需要它**，所以这段在 `#if PTR_SIZE == 4` 里。

## 【常见卡点】

- **`vpop` 可能产码**（浮点栈、跳转链），不是纯指针操作。
- **`vswap`/`vsetc` 遇到 CMP/JMP 先物化**，否则跳转链会被打乱。
- **`gaddrof` 只改标志**。
- **`lvalue_type` 记录左值字节大小**，供 `load`/`store` 选指令宽度。
- **`force_charshort_cast` 用移位实现符号扩展**。
